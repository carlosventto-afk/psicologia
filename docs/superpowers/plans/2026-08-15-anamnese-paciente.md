# Segmento "Anamnese" no cadastro do paciente Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Adicionar uma aba "Anamnese" na página de detalhe do paciente, com 11 campos clínicos de texto livre e um histórico de atualizações (`AnamneseFollowup`) que registra, a cada salvamento, só os campos que mudaram e uma observação livre opcional.

**Architecture:** Duas tabelas novas — `Anamnese` (1:1 com `Paciente`, estado atual) e `AnamneseFollowup` (histórico append-only, `jsonb` com os campos alterados por evento). Uma server action (`salvarAnamnese`) faz o diff entre o valor salvo e o valor novo do formulário antes de fazer o `upsert` e decide se cria um followup. A página de detalhe do paciente ganha abas (Dados/Anamnese/Sessões) via query param, sem estado client-side.

**Tech Stack:** Next.js 16 App Router (Server Components/Actions), Supabase Postgres. Sem framework de teste automatizado neste projeto — verificação via scripts Node ad-hoc com `pg`/`@supabase/supabase-js`, e verificação final no navegador via Playwright CLI contra build+preview.

**Spec:** `docs/superpowers/specs/2026-08-15-anamnese-paciente-design.md`

## Global Constraints

- Os 11 campos da anamnese são todos `text`, nullable, sem default — nenhum é obrigatório.
- `Anamnese` só é criada no primeiro salvamento (não existe uma linha vazia por paciente por padrão).
- `AnamneseFollowup` é append-only: a aplicação nunca faz `update`/`delete` nela, só `insert`.
- Um followup só é criado se `alteracoes.length > 0` **ou** a observação da atualização veio preenchida — salvar sem mudar nada e sem observação não gera evento.
- RLS de ambas as tabelas segue o padrão de `PagamentoSessao` (sem coluna `owner` própria — escopo via join até `Paciente.owner`), já existente em `supabase/migrations/20260727000003_enable_rls_policies.sql`.
- Conexão direta ao Postgres de produção pra migrations/verificação:
  `postgresql://postgres:<SUPABASE_DB_PASSWORD>@db.rohulajgyxdangxfurha.supabase.co:5432/postgres` — `SUPABASE_DB_PASSWORD` já está disponível no ambiente. `.env.local` em `web/` tem `NEXT_PUBLIC_SUPABASE_URL` e `SUPABASE_SERVICE_ROLE_KEY` pros scripts que usam `@supabase/supabase-js`.

```sql
-- Segmento "Anamnese" no cadastro do paciente (backlog item 12): dados
-- clinicos (medicacao, queixa inicial, historico etc.), separados dos
-- dados cadastrais/financeiros que ja existem em Paciente.
--
-- Anamnese guarda o estado ATUAL (1:1 com Paciente, criada so quando o
-- profissional salva pela primeira vez). AnamneseFollowup e um historico
-- append-only: cada edicao gera um evento com data/hora, so os campos que
-- mudaram naquele salvamento (nao duplica os que ficaram iguais) e uma
-- observacao livre opcional -- permite ao profissional acompanhar a
-- evolucao do paciente e consultar informacoes antigas.
create table "Anamnese" (
  id bigint generated by default as identity primary key,
  paciente bigint not null unique references "Paciente"(id) on delete cascade,
  medicacao_em_uso text,
  medico_responsavel text,
  terapia_desde text,
  atendido_desde text,
  queixa_inicial text,
  desenvolvimento_queixa text,
  historico_familiar text,
  tratamento_anterior text,
  uso_substancias text,
  hipotese_diagnostica text,
  expectativas text,
  atualizado_em timestamptz not null default now()
);

-- append-only: a aplicacao nunca faz update/delete aqui, so insert.
create table "AnamneseFollowup" (
  id bigint generated by default as identity primary key,
  anamnese bigint not null references "Anamnese"(id) on delete cascade,
  criado_em timestamptz not null default now(),
  observacao text,
  alteracoes jsonb not null default '[]'::jsonb
);

create index anamnesefollowup_anamnese_idx on "AnamneseFollowup"(anamnese);

alter table "Anamnese" enable row level security;
alter table "AnamneseFollowup" enable row level security;

-- mesmo padrao de PagamentoSessao: sem coluna owner direta, escopo via
-- join ate Paciente.owner.
create policy "anamnese_all_via_paciente" on "Anamnese"
  for all
  using (
    exists (
      select 1 from "Paciente" p
      where p.id = "Anamnese".paciente
        and (p.owner = auth.uid() or public.is_admin())
    )
  )
  with check (
    exists (
      select 1 from "Paciente" p
      where p.id = "Anamnese".paciente
        and (p.owner = auth.uid() or public.is_admin())
    )
  );

create policy "anamnesefollowup_all_via_anamnese" on "AnamneseFollowup"
  for all
  using (
    exists (
      select 1 from "Anamnese" a
      join "Paciente" p on p.id = a.paciente
      where a.id = "AnamneseFollowup".anamnese
        and (p.owner = auth.uid() or public.is_admin())
    )
  )
  with check (
    exists (
      select 1 from "Anamnese" a
      join "Paciente" p on p.id = a.paciente
      where a.id = "AnamneseFollowup".anamnese
        and (p.owner = auth.uid() or public.is_admin())
    )
  );
```

```js
import { normalizarIds, normalizarIdsLista } from "@/lib/normalizar-ids";

export async function buscarAnamnese(pacienteId) {
  const supabase = await createClient();
  const { data, error } = await supabase
    .from("Anamnese")
    .select(
      "id, paciente, medicacao_em_uso, medico_responsavel, terapia_desde, atendido_desde, queixa_inicial, desenvolvimento_queixa, historico_familiar, tratamento_anterior, uso_substancias, hipotese_diagnostica, expectativas, atualizado_em"
```

## Task 4: Server action `salvarAnamnese` (diff + upsert + followup)

**Files:**
- Create: `web/lib/actions/anamnese.js`

**Interfaces:**
- Consumes: `buscarAnamnese` (Task 3), `CAMPOS_ANAMNESE` (Task 2), `createClient` de `@/lib/supabase/server`.
- Produces: `salvarAnamnese(pacienteId, prevState, formData)` — assinatura igual a `atualizarPaciente` em `web/lib/actions/pacientes.js` (compatível com `useActionState`). Retorna `{ error }` em falha; em sucesso faz `redirect` (sem retorno).

- [ ] **Step 1: Escrever a action**

```js
"use server";

import { redirect } from "next/navigation";
import { revalidatePath } from "next/cache";
import { createClient } from "@/lib/supabase/server";
import { buscarAnamnese } from "@/lib/data/anamnese";
import { CAMPOS_ANAMNESE } from "@/lib/anamnese-campos";

function dadosDoFormulario(formData) {
  const dados = {};
  for (const { chave } of CAMPOS_ANAMNESE) {
    dados[chave] = formData.get(chave) || null;
  }
  return dados;
}

function calcularAlteracoes(anamneseAtual, novosValores) {
  const alteracoes = [];
  for (const { chave } of CAMPOS_ANAMNESE) {
    const valorAnterior = anamneseAtual?.[chave] ?? null;
    const valorNovo = novosValores[chave];
    if (valorAnterior !== valorNovo) {
      alteracoes.push({ campo: chave, valor_anterior: valorAnterior, valor_novo: valorNovo });
    }
  }
  return alteracoes;
}

export async function salvarAnamnese(pacienteId, prevState, formData) {
  const anamneseAtual = await buscarAnamnese(pacienteId);
  const novosValores = dadosDoFormulario(formData);
  const alteracoes = calcularAlteracoes(anamneseAtual, novosValores);
  const observacao = formData.get("observacao") || null;

  const supabase = await createClient();

  const { data: anamnese, error: erroUpsert } = await supabase
    .from("Anamnese")
    .upsert(
      { paciente: pacienteId, ...novosValores, atualizado_em: new Date().toISOString() },
      { onConflict: "paciente" }
    )
    .select("id")
    .single();

  if (erroUpsert) {
    return { error: "Não foi possível salvar a anamnese." };
  }

  if (alteracoes.length > 0 || observacao) {
    const { error: erroFollowup } = await supabase
      .from("AnamneseFollowup")
      .insert({ anamnese: anamnese.id, observacao, alteracoes });

    if (erroFollowup) {
      return { error: "Não foi possível salvar a anamnese." };
    }
  }

  revalidatePath(`/pacientes/${pacienteId}`);
  redirect(`/pacientes/${pacienteId}?aba=anamnese`);
}
```
